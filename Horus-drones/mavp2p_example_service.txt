#Example .service file to put into /etc/systemmd/system/ in your pi 2w

[Unit]
Description=mavp2p MAVLink router
After=network.target

[Service]
ExecStart=/usr/local/bin/mavp2p serial:/dev/serial0:921600 udpc:192.168.1.3:14550 udpc:127.0.0.1:14551 #Change the 192.168.* ip to the ip of your desired ground station laptop
Restart=always
RestartSec=5
User=orb1 #Change to your desired user

[Install]
WantedBy=multi-user.target
